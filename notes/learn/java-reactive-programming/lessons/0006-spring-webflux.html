Lesson 06 / 08 Spring WebFlux Integration

Spring WebFlux is the reactive web stack built on Reactor and Netty. It replaces Spring MVC's Tomcat thread-per-request model with an event-loop model. Your Empower PMS API layer lives here. This lesson covers controllers, reactive repositories (R2DBC), and WebClient for outbound HTTP.

Key Dependency
You need spring-boot-starter-webflux — NOT spring-boot-starter-web. Having both on the classpath will default to MVC. Add spring-boot-starter-data-r2dbc and the r2dbc-mssql driver for SQL Server. 
<!-- pom.xml -->
<dependency>
  <groupId>org.springframework.boot</groupId>
  <artifactId>spring-boot-starter-webflux</artifactId>
</dependency>
<dependency>
  <groupId>org.springframework.boot</groupId>
  <artifactId>spring-boot-starter-data-r2dbc</artifactId>
</dependency>
<dependency>
  <groupId>io.r2dbc</groupId>
  <artifactId>r2dbc-mssql</artifactId>
</dependency>

Reactive Controllers

WebFlux controllers look almost identical to MVC controllers — the only difference is the return type: Mono<T> or Flux<T> instead of T.

@RestController @RequestMapping("/api/v1/prescriptions") public class PrescriptionController { private final PrescriptionService svc; // GET single — returns Mono @GetMapping("/{id}") public Mono<ResponseEntity<PrescriptionDTO>> getById(@PathVariable Long id) { return svc.findById(id) .map(ResponseEntity::ok) .defaultIfEmpty(ResponseEntity.notFound().build()); } // GET many — returns Flux (streams as NDJSON or SSE) @GetMapping(produces = MediaType.APPLICATION_NDJSON_VALUE) public Flux<PrescriptionDTO> getByPatient(@RequestParam Long patientId) { return svc.findByPatient(patientId); } // POST — returns Mono of created resource @PostMapping @ResponseStatus(HttpStatus.CREATED) public Mono<PrescriptionDTO> create(@RequestBody Mono<FillRequest> body) { return body.flatMap(svc::fill); } // DELETE — returns Mono<Void> @DeleteMapping("/{id}") @ResponseStatus(HttpStatus.NO_CONTENT) public Mono<Void> cancel(@PathVariable Long id) { return svc.cancel(id); } // SSE endpoint — infinite stream pushed to browser/client @GetMapping(value = "/stream", produces = MediaType.TEXT_EVENT_STREAM_VALUE) public Flux<ServerSentEvent<PrescriptionDTO>> stream() { return svc.liveStream() .map(dto -> ServerSentEvent.<PrescriptionDTO>builder() .event("prescription") .data(dto) .build()); } }
// R2DBC reactive repository — no JDBC, fully non-blocking @Repository public interface PrescriptionR2dbcRepository extends ReactiveCrudRepository<Prescription, Long> { // Derived queries work just like Spring Data JPA Flux<Prescription> findByPatientId(Long patientId); Flux<Prescription> findByDeaScheduleIsNotNull(); Mono<Long> countByStatus(String status); // Custom query with @Query @Query(""" SELECT p.* FROM prescriptions p JOIN patients pt ON p.patient_id = pt.id WHERE pt.facility_code = :facilityCode AND p.status = 'PENDING' ORDER BY p.created_at ASC """) Flux<Prescription> findPendingByFacility(@Param("facilityCode") String code); } // Entity — annotated with Spring Data R2DBC annotations @Table("prescriptions") public class Prescription { @Id private Long id; private Long patientId; private String ndc; private String status; private String deaSchedule; private LocalDateTime createdAt; // Note: NO @OneToMany — R2DBC has no lazy loading. Joins are explicit queries. }
@Service @RequiredArgsConstructor public class PrescriptionService { private final PrescriptionR2dbcRepository rxRepo; private final PatientR2dbcRepository patientRepo; private final FormularyWebClient formulary; public Mono<PrescriptionDTO> findById(Long id) { return rxRepo.findById(id) .flatMap(rx -> Mono.zip( Mono.just(rx), patientRepo.findById(rx.getPatientId()) )) .map(t -> PrescriptionDTO.of(t.getT1(), t.getT2())); } public Flux<PrescriptionDTO> findByPatient(Long patientId) { return patientRepo.findById(patientId) .switchIfEmpty(Mono.error(new NotFoundException("Patient", patientId))) .flatMapMany(p -> rxRepo.findByPatientId(p.getId())) .map(rx -> PrescriptionDTO.of(rx, /* patient already in rx DTO */ null)); } }

WebClient: Reactive Outbound HTTP

For calling the formulary service, insurance verification, or any external API — use WebClient, not RestTemplate. RestTemplate is blocking.

@Configuration
public class WebClientConfig {
    @Bean
    public WebClient formularyWebClient() {
        return WebClient.builder()
            .baseUrl("https://formulary.internal.empower")
            .defaultHeader("X-Service", "empower-pms")
            .build();
    }
}

@Service
public class FormularyWebClient {

    private final WebClient client;

    public Mono<Coverage> getCoverage(Long patientId, String ndc) {
        return client.get()
            .uri("/coverage/{patientId}/{ndc}", patientId, ndc)
            .retrieve()
            .onStatus(HttpStatusCode::is4xxClientError,
                res -> Mono.error(new FormularyException("Coverage not found")))
            .bodyToMono(Coverage.class)
            .retryWhen(Retry.backoff(2, Duration.ofMillis(300))
                .filter(ex -> ex instanceof WebClientRequestException))
            .timeout(Duration.ofSeconds(3));
    }

    // Streaming response from formulary bulk endpoint
    public Flux<DrugInfo> streamDrugCatalog() {
        return client.get()
            .uri("/drugs/catalog")
            .retrieve()
            .bodyToFlux(DrugInfo.class);  // streams NDJSON line by line
    }
}

Global Error Handling with @ControllerAdvice

@RestControllerAdvice
public class ReactiveExceptionHandler {

    @ExceptionHandler(NotFoundException.class)
    @ResponseStatus(HttpStatus.NOT_FOUND)
    public Mono<ErrorResponse> handleNotFound(NotFoundException ex) {
        return Mono.just(new ErrorResponse(ex.getMessage(), 404));
    }

    @ExceptionHandler(WebExchangeBindException.class)
    @ResponseStatus(HttpStatus.BAD_REQUEST)
    public Mono<ErrorResponse> handleValidation(WebExchangeBindException ex) {
        String msg = ex.getBindingResult().getFieldErrors().stream()
            .map(fe -> fe.getField() + ": " + fe.getDefaultMessage())
            .collect(Collectors.joining("; "));
        return Mono.just(new ErrorResponse(msg, 400));
    }
}
