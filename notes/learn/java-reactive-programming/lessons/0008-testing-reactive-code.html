Lesson 08 / 08 Testing Reactive Code

You cannot test reactive code with plain JUnit assertEquals — the stream is asynchronous and nothing executes until subscribed. Reactor ships StepVerifier for unit-testing publishers, and Spring provides WebTestClient for integration testing WebFlux endpoints. This is the last piece you need to ship production reactive code.

StepVerifier: The Reactive Test DSL

StepVerifier subscribes to a publisher and lets you declare expectations step-by-step. It blocks until the stream completes (or times out) — making async code testable synchronously.

// Dependency
<dependency>
  <groupId>io.projectreactor</groupId>
  <artifactId>reactor-test</artifactId>
  <scope>test</scope>
</dependency>

Testing a Mono

@Test
void findById_shouldReturnPatient() {
    // Given
    when(patientRepo.findById(1L))
        .thenReturn(Mono.just(new Patient(1L, "Alice", true)));

    // When + Then
    StepVerifier.create(patientService.findById(1L))
        .assertNext(p -> {
            assertThat(p.getName()).isEqualTo("Alice");
            assertThat(p.isActive()).isTrue();
        })
        .verifyComplete();  // asserts onComplete() fired, no error
}

@Test
void findById_notFound_shouldReturnEmpty() {
    when(patientRepo.findById(99L)).thenReturn(Mono.empty());

    StepVerifier.create(patientService.findById(99L))
        .verifyComplete();  // no items, just completion
}

@Test
void fill_inactivePatient_shouldError() {
    when(patientRepo.findById(1L))
        .thenReturn(Mono.just(new Patient(1L, "Bob", false)));  // inactive!

    StepVerifier.create(fillService.fill(1L, "12345678901", 30))
        .verifyError(InactivePatientException.class);
}

Testing a Flux

@Test
void findByPatient_shouldReturnAllPrescriptions() {
    List<Prescription> rxList = List.of(
        new Prescription(1L, "Lisinopril"),
        new Prescription(2L, "Metformin")
    );
    when(rxRepo.findByPatientId(42L)).thenReturn(Flux.fromIterable(rxList));

    StepVerifier.create(rxService.findByPatient(42L))
        .expectNextMatches(rx -> rx.getDrugName().equals("Lisinopril"))
        .expectNextMatches(rx -> rx.getDrugName().equals("Metformin"))
        .verifyComplete();
}

@Test
void findByPatient_countOnly() {
    when(rxRepo.findByPatientId(42L)).thenReturn(Flux.range(1, 50)
        .map(i -> new Prescription((long) i, "Drug" + i)));

    StepVerifier.create(rxService.findByPatient(42L))
        .expectNextCount(50)
        .verifyComplete();
}

Testing Time-Based Operations

For retries, timeouts, and delays — use StepVerifier.withVirtualTime() to skip clock time entirely:

@Test
void retryWith3Attempts_onTransientFailure() {
    AtomicInteger attempts = new AtomicInteger(0);

    Mono<Void> sendWithRetry = Mono.fromRunnable(() -> {
        if (attempts.incrementAndGet() < 3)
            throw new TransientServiceBusException("transient");
    }).retryWhen(Retry.backoff(3, Duration.ofSeconds(1)));

    StepVerifier.withVirtualTime(() -> sendWithRetry)
        .thenAwait(Duration.ofSeconds(10))  // advance virtual clock
        .verifyComplete();

    assertThat(attempts.get()).isEqualTo(3);
}

@Test
void timeout_shouldErrorAfter3Seconds() {
    Mono<Coverage> slowFormulary = Mono.never()  // never completes
        .timeout(Duration.ofSeconds(3));

    StepVerifier.withVirtualTime(() -> slowFormulary)
        .thenAwait(Duration.ofSeconds(4))
        .verifyError(TimeoutException.class);
}

WebTestClient: Integration Testing WebFlux Endpoints

@SpringBootTest(webEnvironment = SpringBootTest.WebEnvironment.RANDOM_PORT)
class PrescriptionControllerIT {

    @Autowired
    private WebTestClient webClient;

    @Test
    void getById_shouldReturn200() {
        webClient.get().uri("/api/v1/prescriptions/1")
            .exchange()
            .expectStatus().isOk()
            .expectBody(PrescriptionDTO.class)
            .value(dto -> assertThat(dto.getId()).isEqualTo(1L));
    }

    @Test
    void getByPatient_shouldReturnFlux() {
        webClient.get().uri("/api/v1/prescriptions?patientId=42")
            .accept(MediaType.APPLICATION_NDJSON)
            .exchange()
            .expectStatus().isOk()
            .expectBodyList(PrescriptionDTO.class)
            .hasSize(3);
    }

    @Test
    void create_shouldReturn201() {
        FillRequest req = new FillRequest(42L, "00071015523", 30);

        webClient.post().uri("/api/v1/prescriptions")
            .bodyValue(req)
            .exchange()
            .expectStatus().isCreated()
            .expectBody(PrescriptionDTO.class)
            .value(dto -> {
                assertThat(dto.getNdc()).isEqualTo("00071015523");
                assertThat(dto.getStatus()).isEqualTo("FILLED");
            });
    }
}

Mocking Reactive Repos with Mockito

@ExtendWith(MockitoExtension.class)
class PrescriptionServiceTest {

    @Mock PrescriptionR2dbcRepository rxRepo;
    @Mock PatientR2dbcRepository     patientRepo;
    @InjectMocks PrescriptionService svc;

    @Test
    void cancel_shouldSetStatusCancelled() {
        Prescription rx = new Prescription(1L, "PENDING");
        when(rxRepo.findById(1L)).thenReturn(Mono.just(rx));
        when(rxRepo.save(any())).thenAnswer(inv -> Mono.just(inv.getArgument(0)));

        StepVerifier.create(svc.cancel(1L))
            .verifyComplete();

        verify(rxRepo).save(argThat(saved ->
            "CANCELLED".equals(saved.getStatus())));
    }
}
